cmake_minimum_required(VERSION 3.22)
project(libporthome C)
include(GNUInstallDirs)
include(GenerateExportHeader)
include(CMakePackageConfigHelpers)
if(NOT COMMAND declare_component_version)
  include("${CMAKE_CURRENT_LIST_DIR}/../cmake/VersionHelpers.cmake")
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  include(CTest)
  find_package(nostrc REQUIRED COMPONENTS blossom)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(PORTHOME_NOSTR REQUIRED IMPORTED_TARGET nostr)
  add_library(nostr INTERFACE IMPORTED)
  set_property(TARGET nostr PROPERTY INTERFACE_LINK_LIBRARIES PkgConfig::PORTHOME_NOSTR)
  find_library(PORTHOME_NIP44_CORE NAMES nostr_nip44_core REQUIRED)
  add_library(nostr_nip44_core UNKNOWN IMPORTED)
  set_target_properties(nostr_nip44_core PROPERTIES IMPORTED_LOCATION "${PORTHOME_NIP44_CORE}")
endif()
if(NOT TARGET nostr_nip44_core)
  message(FATAL_ERROR "libporthome requires nip44. Set ENABLE_NIP44=ON or BUILD_LIBPORTHOME=OFF.")
endif()
if(NOT TARGET nostrc::blossom)
  message(FATAL_ERROR "libporthome requires libblossom. Set BUILD_LIBBLOSSOM=ON or BUILD_LIBPORTHOME=OFF.")
endif()
find_package(OpenSSL REQUIRED)
add_library(porthome src/crypto.c src/manifest.c src/cbor.c src/wrapkey.c src/index.c)
add_library(nostrc::porthome ALIAS porthome)
target_compile_features(porthome PUBLIC c_std_11)
target_compile_definitions(porthome PRIVATE _POSIX_C_SOURCE=200809L)
target_compile_options(porthome PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation)
target_include_directories(porthome
  PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
         $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/include>
         $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
  PRIVATE src)
target_link_libraries(porthome PRIVATE nostrc::blossom nostr nostr_nip44_core OpenSSL::Crypto)
generate_export_header(porthome EXPORT_MACRO_NAME PORTHOME_API
  EXPORT_FILE_NAME "${CMAKE_CURRENT_BINARY_DIR}/include/porthome/export.h")
if(NOT BUILD_SHARED_LIBS)
  target_compile_definitions(porthome PUBLIC PORTHOME_STATIC_DEFINE)
endif()
set_target_properties(porthome PROPERTIES C_VISIBILITY_PRESET hidden)
declare_component_version(porthome 0 1 0)
set_target_properties(porthome PROPERTIES
  COMPONENT_VERSION_HEADER_INSTALL_DIR "${CMAKE_INSTALL_INCLUDEDIR}/porthome")
apply_versioning(porthome)
if(COMMAND apply_sanitizers)
  apply_sanitizers(porthome)
endif()
configure_file(porthome.pc.in porthome.pc @ONLY)
configure_package_config_file("${CMAKE_CURRENT_LIST_DIR}/../cmake/nostrcConfig.cmake.in"
  "${CMAKE_CURRENT_BINARY_DIR}/nostrcConfig.cmake"
  INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/nostrc")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/nostrcConfigVersion.cmake"
  VERSION "${porthome_VERSION}" COMPATIBILITY SameMajorVersion)
install(TARGETS porthome ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/porthome DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/include/porthome/export.h"
  DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/porthome)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/porthome.pc" DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/nostrcConfig.cmake"
  "${CMAKE_CURRENT_BINARY_DIR}/nostrcConfigVersion.cmake"
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/nostrc)
if(BUILD_TESTING)
  add_executable(test-porthome-api tests/test_api.c)
  target_link_libraries(test-porthome-api PRIVATE nostrc::porthome)
  target_compile_options(test-porthome-api PRIVATE -Wall -Wextra -Werror)
  if(COMMAND apply_sanitizers)
    apply_sanitizers(test-porthome-api)
  endif()
  add_test(NAME test-porthome-api COMMAND test-porthome-api)
  set_tests_properties(test-porthome-api PROPERTIES TIMEOUT 30 LABELS "libporthome;unit")
endif()
