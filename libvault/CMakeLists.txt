cmake_minimum_required(VERSION 3.22)
project(libvault C)
include(GNUInstallDirs)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/NostrcVisibility.cmake")
include(CMakePackageConfigHelpers)
if(NOT COMMAND declare_component_version)
  include("${CMAKE_CURRENT_LIST_DIR}/../cmake/VersionHelpers.cmake")
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  include(CTest)
  find_package(nostrc REQUIRED COMPONENTS porthome blossom)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(VAULT_NOSTR REQUIRED IMPORTED_TARGET nostr)
  add_library(nostr INTERFACE IMPORTED)
  set_property(TARGET nostr PROPERTY INTERFACE_LINK_LIBRARIES PkgConfig::VAULT_NOSTR)
  pkg_check_modules(VAULT_NIP59 REQUIRED IMPORTED_TARGET nip59)
  add_library(nip59 INTERFACE IMPORTED)
  set_property(TARGET nip59 PROPERTY INTERFACE_LINK_LIBRARIES PkgConfig::VAULT_NIP59)
  find_library(VAULT_NIP44_CORE NAMES nostr_nip44_core REQUIRED)
  add_library(nostr_nip44_core UNKNOWN IMPORTED)
  set_target_properties(nostr_nip44_core PROPERTIES IMPORTED_LOCATION "${VAULT_NIP44_CORE}")
endif()
if(NOT TARGET nostrc::porthome)
  message(FATAL_ERROR "libvault requires libporthome. Set BUILD_LIBPORTHOME=ON or BUILD_LIBVAULT=OFF.")
endif()
if(NOT TARGET nostrc::blossom)
  message(FATAL_ERROR "libvault requires libblossom. Set BUILD_LIBBLOSSOM=ON or BUILD_LIBVAULT=OFF.")
endif()
if(NOT TARGET nostr_nip44_core)
  message(FATAL_ERROR "libvault requires nip44. Set ENABLE_NIP44=ON or BUILD_LIBVAULT=OFF.")
endif()
if(NOT TARGET nip59)
  message(FATAL_ERROR "libvault requires nip59. Set ENABLE_NIP59=ON or BUILD_LIBVAULT=OFF.")
endif()
find_package(OpenSSL REQUIRED)
add_library(vault src/ring.c src/revoke.c src/reseal.c src/snapshot.c src/fold.c src/json.c src/codec.c src/vsp.c src/nip59.c src/tree.c src/fsck.c src/fetch.c src/parents.c src/nip94.c)
add_library(nostrc::vault ALIAS vault)
target_compile_features(vault PUBLIC c_std_11)
target_compile_definitions(vault PRIVATE _POSIX_C_SOURCE=200809L)
target_compile_options(vault PRIVATE -Wall -Wextra -Werror)
target_include_directories(vault PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src)
target_link_libraries(vault PUBLIC nostrc::porthome nostrc::blossom PRIVATE nostr nostr_nip44_core nip59 OpenSSL::Crypto)
nostrc_generate_export_header(vault VAULT_API vault/export.h)
set_target_properties(vault PROPERTIES C_VISIBILITY_PRESET hidden)
declare_component_version(vault 0 7 0)
set_target_properties(vault PROPERTIES COMPONENT_VERSION_HEADER_INSTALL_DIR "${CMAKE_INSTALL_INCLUDEDIR}/vault")
apply_versioning(vault)
if(COMMAND apply_sanitizers)
  apply_sanitizers(vault)
endif()
configure_file(vault.pc.in vault.pc @ONLY)
configure_package_config_file("${CMAKE_CURRENT_LIST_DIR}/../cmake/nostrcConfig.cmake.in"
  "${CMAKE_CURRENT_BINARY_DIR}/nostrcConfig.cmake" INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/nostrc")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/nostrcConfigVersion.cmake" VERSION "${vault_VERSION}" COMPATIBILITY SameMajorVersion)
install(TARGETS vault ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/vault DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/include/vault/export.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/vault)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/vault.pc" DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/nostrcConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/nostrcConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/nostrc)
if(BUILD_TESTING)
  foreach(name ring_derivation revoke_lazy fold_frontier codecs grant_nip59 grant_nip4e tree_ops nip94_export)
    add_executable(test_vault_${name} tests/test_${name}.c)
    target_include_directories(test_vault_${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../tests/common)
    target_compile_options(test_vault_${name} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_compile_definitions(test_vault_${name} PRIVATE _POSIX_C_SOURCE=200809L)
    target_link_libraries(test_vault_${name} PRIVATE nostrc::vault nostr nostr_nip44_core)
    if(name MATCHES "^grant_nip(59|4e)$")
      target_link_libraries(test_vault_${name} PRIVATE nip59)
    endif()
    if(name STREQUAL "nip94_export")
      # Decrypts with only the returned tags and re-derives chunk keys locally.
      find_package(Threads REQUIRED)
      target_link_libraries(test_vault_${name} PRIVATE OpenSSL::Crypto Threads::Threads)
    endif()
    if(COMMAND apply_sanitizers)
      apply_sanitizers(test_vault_${name})
    endif()
    add_test(NAME test_vault_${name} COMMAND test_vault_${name})
    set_tests_properties(test_vault_${name} PROPERTIES TIMEOUT 30 LABELS "libvault;unit")
  endforeach()
endif()
