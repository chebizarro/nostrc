cmake_minimum_required(VERSION 3.22)
project(libvault_marmot C)
include(GNUInstallDirs)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/NostrcVisibility.cmake")
include(CMakePackageConfigHelpers)
if(NOT COMMAND declare_component_version)
  include("${CMAKE_CURRENT_LIST_DIR}/../cmake/VersionHelpers.cmake")
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  include(CTest)
  find_package(nostrc REQUIRED COMPONENTS vault)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(VAULT_MARMOT_MARMOT REQUIRED IMPORTED_TARGET "marmot>=0.13.0")
  add_library(nostrc::marmot INTERFACE IMPORTED)
  set_property(TARGET nostrc::marmot PROPERTY INTERFACE_LINK_LIBRARIES PkgConfig::VAULT_MARMOT_MARMOT)
  pkg_check_modules(VAULT_MARMOT_NOSTR REQUIRED IMPORTED_TARGET nostr)
  add_library(nostr INTERFACE IMPORTED)
  set_property(TARGET nostr PROPERTY INTERFACE_LINK_LIBRARIES PkgConfig::VAULT_MARMOT_NOSTR)
endif()
if(NOT TARGET nostrc::vault)
  message(FATAL_ERROR "libvault-marmot requires libvault. Set BUILD_LIBVAULT=ON or BUILD_LIBVAULT_MARMOT=OFF.")
endif()
if(NOT TARGET nostrc::marmot)
  if(NOT TARGET marmot)
    message(FATAL_ERROR "libvault-marmot requires libmarmot. Set BUILD_LIBMARMOT=ON or BUILD_LIBVAULT_MARMOT=OFF.")
  endif()
  add_library(nostrc::marmot ALIAS marmot)
endif()
find_package(OpenSSL REQUIRED)
add_library(vault_marmot src/carrier.c src/seed.c)
add_library(nostrc::vault-marmot ALIAS vault_marmot)
target_compile_features(vault_marmot PUBLIC c_std_11)
target_compile_options(vault_marmot PRIVATE -Wall -Wextra -Werror)
target_include_directories(vault_marmot PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src)
target_link_libraries(vault_marmot PUBLIC nostrc::vault nostrc::marmot PRIVATE nostr OpenSSL::Crypto)
nostrc_generate_export_header(vault_marmot VAULT_MARMOT_API vault-marmot/export.h)
set_target_properties(vault_marmot PROPERTIES C_VISIBILITY_PRESET hidden OUTPUT_NAME vault-marmot)
declare_component_version(vault_marmot 0 1 1)
set_target_properties(vault_marmot PROPERTIES COMPONENT_VERSION_HEADER_INSTALL_DIR "${CMAKE_INSTALL_INCLUDEDIR}/vault-marmot")
apply_versioning(vault_marmot)
if(COMMAND apply_sanitizers)
  apply_sanitizers(vault_marmot)
endif()
# libmarmot's historical .pc omits some private static dependencies.
set(VAULT_MARMOT_PC_PRIVATE_REQUIRES "nostr openssl libsecp256k1")
if(SQLITE3_FOUND)
  string(APPEND VAULT_MARMOT_PC_PRIVATE_REQUIRES " sqlite3")
endif()
if(LMDB_FOUND)
  string(APPEND VAULT_MARMOT_PC_PRIVATE_REQUIRES " lmdb")
endif()
configure_file(vault-marmot.pc.in vault-marmot.pc @ONLY)
configure_package_config_file("${CMAKE_CURRENT_LIST_DIR}/../cmake/nostrcConfig.cmake.in"
  "${CMAKE_CURRENT_BINARY_DIR}/nostrcConfig.cmake" INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/nostrc")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/nostrcConfigVersion.cmake"
  VERSION "${vault_marmot_VERSION}" COMPATIBILITY SameMajorVersion)
install(TARGETS vault_marmot ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/vault-marmot DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/include/vault-marmot/export.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/vault-marmot)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/vault-marmot.pc" DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/nostrcConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/nostrcConfigVersion.cmake"
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/nostrc)
if(BUILD_TESTING)
  add_executable(test_vault_grant_marmot tests/test_grant_marmot.c)
  target_compile_options(test_vault_grant_marmot PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_compile_definitions(test_vault_grant_marmot PRIVATE _POSIX_C_SOURCE=200809L)
  target_link_libraries(test_vault_grant_marmot PRIVATE nostrc::vault-marmot nostr)
  if(COMMAND apply_sanitizers)
    apply_sanitizers(test_vault_grant_marmot)
  endif()
  add_test(NAME test_vault_grant_marmot COMMAND test_vault_grant_marmot)
  set_tests_properties(test_vault_grant_marmot PROPERTIES TIMEOUT 60 LABELS "libvault-marmot;unit")
endif()
