# Layer-7 assembly; no GLib or Marmot dependency.
if(NOT TARGET nostrc::vault OR NOT TARGET nip42 OR NOT TARGET nip11)
  message(FATAL_ERROR "vaultctl requires BUILD_LIBVAULT, ENABLE_NIP42 and ENABLE_NIP11")
endif()
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
add_executable(vaultctl main.c state.c import.c blobs.c relay.c records.c commands.c)
target_compile_features(vaultctl PRIVATE c_std_11)
target_compile_definitions(vaultctl PRIVATE _POSIX_C_SOURCE=200809L _DARWIN_C_SOURCE _DEFAULT_SOURCE)
target_compile_options(vaultctl PRIVATE -Wall -Wextra -Werror)
target_include_directories(vaultctl PRIVATE ${CMAKE_SOURCE_DIR}/nips/nip42/include ${CMAKE_SOURCE_DIR}/nips/nip11/include)
target_link_libraries(vaultctl PRIVATE nostrc::vault nostrc::porthome nostrc::blossom nostr libgo nip42 nip11 OpenSSL::Crypto CURL::libcurl Threads::Threads)
declare_component_version(vaultctl 0 3 1)
if(COMMAND apply_sanitizers)
  apply_sanitizers(vaultctl)
endif()
install(TARGETS vaultctl RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_executable(vaultctl_fixture_helper tests/fixture_helper.c state.c)
  target_compile_definitions(vaultctl_fixture_helper PRIVATE _POSIX_C_SOURCE=200809L _DARWIN_C_SOURCE _DEFAULT_SOURCE)
  target_compile_options(vaultctl_fixture_helper PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_link_libraries(vaultctl_fixture_helper PRIVATE nostrc::vault nostr libgo OpenSSL::Crypto)
  if(COMMAND apply_sanitizers)
    apply_sanitizers(vaultctl_fixture_helper)
  endif()
  add_test(NAME test_vaultctl_fixtures COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures.py $<TARGET_FILE:vaultctl> $<TARGET_FILE:vaultctl_fixture_helper>)
  set_tests_properties(test_vaultctl_fixtures PROPERTIES TIMEOUT 120 LABELS "vaultctl;integration" RESOURCE_LOCK vaultctl_fixture)
endif()
